def gv

pipeline {
    agent any

    parameters {
        choice(name: 'VERSION', choices: ['v1.0', 'v1.1', 'v1.2'], description: 'A parameter for the pipeline')
        booleanParam(name: 'executeTests', defaultValue: true, description: 'Execute tests?')
    }

    stages {
        stage("init") {
            steps {
                script {
                    gv = load "script.groovy"
                }
            }
        }

        stage("build") {
            steps {
                script {
                    gv.buildApp()
                }
            }
        }

        stage("test") {
            when {
                expression {
                    params.executeTests == true   // explicit boolean check
                }
            }
            steps {
                script {
                    gv.testApp()
                }
            }
        }

        stage("deploy") {
            steps {
                script {
                    env.ENV = input(
                        message: 'Select the environment to deploy to',
                        ok: 'Done',                                   // was: ok "Done"
                        parameters: [
                            choice(name: 'ENV', choices: ['dev', 'staging', 'prod'], description: '')
                        ]
                    )
                    gv.deployApp()
                    echo "Deploying to ${env.ENV}"                   // was: ${ENV}
                }
            }
        }
    }
}
