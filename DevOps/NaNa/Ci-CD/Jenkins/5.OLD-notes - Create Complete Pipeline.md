3.add git repo and git cred
4.pre configure git repo and registry creds in jenkins cred
5.create pipline as below in jenkins
6.build
pipeline {
    agent any
    environment{
        REGISTRY = 'git.example.com'
        IMAGE_NAME = 'robot-market/jenkins'
        IMAGE_TAG = "${env.BUILD_NUMBER}"
    }
    stages {
        stage('Build docker image') {
            steps{
                script{
                    echo "building the docker image"
                    sh 'docker build -t ${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG} .'
                }
            }

        }
        stage('Push docker image') {
            steps{
                withCredentials([usernamePassword(
                credentialsId: 'Gitea-Docker',
                 usernameVariable: 'USER',
                  passwordVariable: 'PASS'
                  )]){
                    echo "pushing the docker image"
                    sh '''
                    echo $PASS | docker login $REGISTRY -u $USER --password-stdin
                    docker push ${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}
                    '''
                }
            }

        }
    }
    post{
    always{
        sh 'docker logout ${REGISTRY} || true'
        sh 'docker rmi ${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG} || true'
    }
    }
}
